def encode(x):return json.dumps(x,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()
def digest(x):return hashlib.sha256(encode(x)).hexdigest()
def read(p):return json.loads(p.read_bytes())
def write(p,x):p.write_bytes(encode(x)+b'\n')
def snapshot(p):return{str(f.relative_to(p)):f.read_bytes()for f in p.rglob('*')if f.is_file()}
def source(tmp_path,name='nano-direct-01',dirname='source'):
 p=tmp_path/dirname;shutil.copytree(BASE/name,p);return p

def build(src,tmp_path,destination=None,mapping=None):
 dest=destination or tmp_path/'packet'
 packet=build_operator_packet(mapping or {'case-a':src},dest,protocol=PROTOCOL,assignment=ASSIGN)
 return dest,packet

def answer(response,index=0,axis=0,judgment='yes',ref=None):
 row=response['cases'][index]['axes'][axis]
 row.update(judgment=judgment,unassessed_reason='reviewer_abstention' if judgment=='unassessed' else None,rationale='Synthetic manual response; no correctness assertion.')
 row['evidence_refs']=[ref or {'document':'evidence','pointer':'/completion'}]

def submit(packet,tmp_path,response=None,raw=None,dest=None):
 p=tmp_path/'response.json';p.write_bytes(raw if raw is not None else encode(response))
 return import_operator_response(packet/'manifest.json',p,dest or tmp_path/'receipt')

def rehash(root):
 p=root/'manifest.json';m=read(p);m['files']={k:hashlib.sha256(v).hexdigest()for k,v in snapshot(root).items()if k!='manifest.json'}
 if 'manifest_sha256'in m:
  del m['manifest_sha256'];m['manifest_sha256']=digest(m)
 write(p,m)

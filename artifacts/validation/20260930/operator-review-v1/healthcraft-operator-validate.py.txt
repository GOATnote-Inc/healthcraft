"""Freeze the operator candidate, run inventoried lint, optionally full tests."""
import hashlib
import json
from pathlib import Path
import shutil
import subprocess
import sys
import tempfile

ROOT = Path('/Users/kiteboard/healthcraft')
OUT = Path('/private/tmp/healthcraft-operator-final')
NAMES = [
    'src/healthcraft/operator_review.py',
    'src/healthcraft/operator_review_report.py',
    'scripts/operator_review.py',
    'tests/test_operator_review.py',
    'tests/test_operator_review_report.py',
    'tests/test_scripts/test_operator_review_cli.py',
    'docs/OPERATOR_REVIEW.md',
    'docs/SYNTHETIC_RECONCILIATION.md',
    'docs/RELEASE_EVIDENCE_PLAN.md',
    'README.md',
]

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def save(name, value):
    with (OUT / name).open('x') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')

mode = sys.argv[1]
if mode == 'lint':
    OUT.mkdir(exist_ok=False)
    save('candidate-inputs.json', {name: digest(ROOT / name) for name in NAMES})
    prior = ROOT / 'artifacts/validation/20260930/grading-source-context-v2/validation/lint-inputs.json'
    names = sorted(set(json.loads(prior.read_text())) | set(NAMES))
    inventory = {name: digest(ROOT / name) for name in names}
    save('lint-inputs.json', inventory)
    snapshot = Path(tempfile.mkdtemp(prefix='healthcraft-operator-lint-', dir='/private/tmp'))
    for name in names:
        destination = snapshot / name
        destination.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(ROOT / name, destination)
        assert digest(destination) == inventory[name]
    command = ['make', 'lint', 'PYTHON=' + str(ROOT / '.venv/bin/python')]
    codes = {}
    for label, cwd in [('root', ROOT), ('isolated', snapshot)]:
        with (OUT / (label + '-make-lint.log')).open('x') as log:
            codes[label] = subprocess.run(command, cwd=cwd, stdout=log, stderr=subprocess.STDOUT).returncode
    receipt = {'command': command, 'snapshot': str(snapshot), 'inputs':len(names),
               'python_files':sum(n.endswith('.py') for n in names), 'exit_codes':codes,
               'excluded':'Immutable result/artifact collections and unrelated untracked directories; every input inventoried.'}
    save('lint-receipt.json', receipt)
    assert all(digest(ROOT / name) == value for name,value in inventory.items())
    print(json.dumps(receipt, indent=2))
    raise SystemExit(codes['isolated'])
elif mode == 'test':
    inventory = json.loads((OUT / 'candidate-inputs.json').read_text())
    assert all(digest(ROOT / name) == value for name,value in inventory.items())
    command = ['make', 'test']
    with (OUT / 'full-suite.log').open('x') as log:
        code = subprocess.run(command, cwd=ROOT, stdout=log, stderr=subprocess.STDOUT).returncode
    assert all(digest(ROOT / name) == value for name,value in inventory.items())
    receipt = {'command':command, 'exit_code':code, 'log_sha256':digest(OUT / 'full-suite.log'),
               'candidate_inputs':inventory, 'python':subprocess.check_output([str(ROOT / '.venv/bin/python'), '--version'], text=True).strip()}
    save('full-suite-receipt.json', receipt)
    print(json.dumps(receipt, indent=2))
    raise SystemExit(code)
else:
    raise ValueError(mode)

@pytest.mark.parametrize('label,response,mode',[
 ('native-assisted','coordinator/synthetic-operator-response.json','incident'),
 ('native-raw','native-raw/public/response-template.json','incident'),
 ('model-assisted','model-assisted/public/response-template.json','incident'),
 ('synthetic-reviewer-a','coordinator/synthetic-reviewer-a-response.json','adjudication'),
 ('synthetic-reviewer-b','coordinator/synthetic-reviewer-b-response.json','adjudication'),
 ('synthetic-resolver','coordinator/resolver-response.json','adjudication')])
def test_actual_issued_responses_roundtrip(label,response,mode):
 pair=(mode,tuple(json.loads((ART/label/'public'/name).read_text()) for name in ('packet.json','response-template.json')))
 value=json.loads((ART/response).read_text())
 expression=drafts.export_expression(pair)
 result=drafts.run_js(pair,value,"(()=>{const initial="+expression+";HealthcraftDrafts.preview(raw);const previewChanges=mutationCount();HealthcraftDrafts.apply();const resumed="+expression+";HealthcraftDrafts.undo();return{initial,resumed,undone:"+expression+",previewChanges};})()")
 assert 'error' not in result,result
 assert result['result']['previewChanges']==0
 assert result['result']['resumed']==value
 assert result['result']['initial']==result['result']['undone']
 if mode=='incident':

  validate_incident_response(result['result']['resumed'],pair[1][0])
 else:
  from healthcraft.operator_adjudication import validate_adjudication_response
  validate_adjudication_response(result['result']['resumed'],pair[1][0])

def source_open(pair, refs):
 page=examples.Page(render_workbench(*pair))
 harness=r'''const fs=require('node:fs'),vm=require('node:vm'),p=JSON.parse(fs.readFileSync(0,'utf8'));const nodes={};function el(tag){return{tagName:tag,children:[],listeners:{},textContent:'',value:'',hidden:false,appendChild(n){this.children.push(n);},replaceChildren(...n){this.children=n;},addEventListener(k,f){this.listeners[k]=f;}};}for(const[tag,a]of p.tags)if(a.id)nodes[a.id]=el(tag);nodes['incident-packet'].textContent=p.packet_text;nodes['incident-template'].textContent=p.template_text;const c={document:{getElementById(i){return nodes[i];},querySelectorAll(){return[];},createElement:el}};vm.createContext(c);vm.runInContext(p.base,c);vm.runInContext(p.script,c);const results=p.refs.map(r=>{const result=c.HealthcraftSources.open(0,r);return{result,display:nodes['source-0-value'].textContent,citation:nodes['source-0-citation'].value,raw:nodes['source-0-raw'].textContent};});process.stdout.write(JSON.stringify(results));'''
 completed=subprocess.run(['node','-e',harness],input=json.dumps({'tags':page.tags,'packet_text':page.scripts['incident-packet'],'template_text':page.scripts['incident-template'],'base':page.scripts['incident-app'],'script':page.scripts['workbench-sources'],'refs':refs}),text=True,capture_output=True,check=True)
 return json.loads(completed.stdout)


def test_response_decoding_cannot_offer_mismatched_raw_citation():
 pair=examples.adjudication.__wrapped__(examples.incident.__wrapped__())
 pair[0]['cases'][0]['documents']['response']['incident_assessment']['summary']='{"result":"nested claim"}'
 pair[1]['packet_sha256']=examples.digest(pair[0])
 rows=source_open(pair,[{'document':'response','pointer':'/incident_assessment/summary'},{'document':'response','pointer':'/incident_assessment/summary','decoded_json_pointer':'/result'}])
 assert rows[0]['display']=='{"result":"nested claim"}'
 assert rows[0]['citation']=='"/incident_assessment/summary"'
 assert rows[1]['result']['status']=='unavailable'
 assert rows[1]['display']==rows[1]['citation']==''

def test_number_text_retains_lexeme_and_null_is_not_unavailable():
 pair=examples.incident.__wrapped__();docs=pair[0]['cases'][0]['documents']
 docs['evidence']['literal']='{"integer":900719925474099312345678901234567890,"decimal":0.10000000000000000000000001,"null":null,"negative_zero":-0.0}'
 pair[0]['cases'][0]['availability']['evidence']['sha256']=examples.digest(docs['evidence']);pair[1]['packet_sha256']=examples.digest(pair[0])
 rows=source_open(pair,[{'document':'evidence','pointer':'/literal','decoded_json_pointer':'/'+key}for key in ('integer','decimal','null','negative_zero')])
 assert [r['display']for r in rows]==['900719925474099312345678901234567890','0.10000000000000000000000001','null','-0.0']
 assert all(r['result']['status']=='available'and r['citation']for r in rows)

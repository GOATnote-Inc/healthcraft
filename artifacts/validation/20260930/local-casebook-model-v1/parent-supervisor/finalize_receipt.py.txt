'red_controls':[{'log':'red.log','failed':28,'reason':'supervisor module absent before implementation'}, {'log':'journal-red.log','failed':3,'passed':29,'reason':'success accepted missing, truncated or duplicate-key mandatory model capture'}, {'log':'partial-frame-red.log','failed':1,'reason':'recv_bytes blocked parent beyond configured deadline; external process-group bound terminated test'}, {'log':'blocked-reply-red.log','failed':1,'reason':'send blocked parent after actual native write; external process-group bound terminated test'}],
'final_own_suite':{'passed':34,'log':'green-final-own.log'},
'lint':{'exit_code':lint.returncode,'format_exit_code':fmt.returncode,'log':'lint.log'},
'limits':['All worker tests are scripted or native-client fake envelopes; zero actual model or hosted requests.','Completion proves worker termination and consistent captured identities, not mechanical or clinical success.','Model-call counts are durable native request attempts, not guaranteed server inference executions.','The parent controls native actions; I/O-only daemon threads bound partial receive frames and blocked replies without later dispatch.','The process deadline does not cancel an already dispatched Ollama daemon request.','Native handlers execute synchronously in the parent with deadline checks before and after; this is not a general arbitrary-code supervisor.','Child receives public instruction/schema/config only; process separation is not a hostile-code filesystem sandbox.','172-test counts overlap 34 owned tests and peer-owned compatibility tests; counts across runtime repetitions are not independent features.','Expected pytest warning: asyncio_mode option unknown with plugin autoload disabled.'],
'files_sha256':{p.name:sha(p) for p in sorted(out.iterdir()) if p.is_file() and p.name!='receipt.json'},
}

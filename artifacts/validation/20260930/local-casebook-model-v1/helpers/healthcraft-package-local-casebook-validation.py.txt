import hashlib,json

ROOT=Path('/Users/kiteboard/healthcraft')

OUT=ROOT/'artifacts/validation/20260930/local-casebook-model-v1'
INDEX=TMP/'healthcraft-local-casebook-validation-index/inventory.json'

index=json.loads(INDEX.read_text())
final=TMP/'healthcraft-local-casebook-final'
full=json.loads((final/'full-suite-receipt.json').read_text())

def copy(src,rel,expected=None):
 src=Path(src);dst=OUT/rel
 assert src.is_file() and not src.is_symlink(),str(src)
 raw=src.read_bytes(); h=hashlib.sha256(raw).hexdigest()
 if expected is not None:assert h==expected,(str(src),h,expected)

 dst.parent.mkdir(parents=True,exist_ok=True)
 with dst.open('xb') as f:f.write(raw)
 assert sha(dst)==h and sha(src)==h
 copies[rel]={'source':str(src),'sha256':h,'bytes':len(raw)}
for item in index['files']:copy(item['source'],item['destination'],item['sha256'])
extras=[
(final/'full-suite.log','final-validation/full-suite.log'),

]
for src,rel in extras:copy(src,rel)

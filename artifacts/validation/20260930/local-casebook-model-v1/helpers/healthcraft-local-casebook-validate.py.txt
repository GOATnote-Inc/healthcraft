"""Freeze this development candidate and retain lint/full-suite exit evidence."""
import hashlib
import json
from pathlib import Path
import shutil
import subprocess
import sys
import tempfile

ROOT = Path('/Users/kiteboard/healthcraft')
OUT = Path('/private/tmp/healthcraft-local-casebook-final')
NAMES = ['src/healthcraft/llm/local_models.py', 'src/healthcraft/reconciliation/controller.py', 'src/healthcraft/reconciliation/public_case.py', 'src/healthcraft/reconciliation/model_worker.py', 'src/healthcraft/reconciliation/model_case.py', 'src/healthcraft/reconciliation/model_cohort.py', 'scripts/reconciliation_local_cohort.py', 'tests/test_llm/test_local_native_json.py', 'tests/test_reconciliation/test_public_case.py', 'tests/test_reconciliation/test_model_worker.py', 'tests/test_reconciliation/test_model_case.py', 'tests/test_reconciliation/test_model_cohort.py', 'configs/evaluation/reconciliation_model_v2/local-pilot-plan.json', 'docs/LOCAL_MODELS.md', 'docs/RECONCILIATION_CASEBOOK.md', 'docs/RECONCILIATION_LOCAL_MODELS.md']

def digest(path): return hashlib.sha256(path.read_bytes()).hexdigest()
def save(name, value):
    with (OUT/name).open('x') as stream:
        json.dump(value, stream, indent=2, allow_nan=False)
        stream.write('\n')

mode=sys.argv[1]
if mode=='lint':
    OUT.mkdir(exist_ok=False)
    candidate={name:digest(ROOT/name) for name in NAMES}
    save('candidate-inputs.json',candidate)
    prior=ROOT/'artifacts/validation/20260930/reconciliation-casebook-v2/validation/lint-inputs.json'
    names=sorted(set(json.loads(prior.read_text()))|set(NAMES))
    inventory={name:digest(ROOT/name) for name in names}
    save('lint-inputs.json',inventory)
    snapshot=Path(tempfile.mkdtemp(prefix='healthcraft-local-casebook-lint-',dir='/private/tmp'))
    for name in names:
        dest=snapshot/name
        dest.parent.mkdir(parents=True,exist_ok=True)
        shutil.copyfile(ROOT/name,dest)
        assert digest(dest)==inventory[name]
    command=['make','lint','PYTHON='+str(ROOT/'.venv/bin/python')]
    codes={}
    for label,cwd in [('root',ROOT),('isolated',snapshot)]:
        with (OUT/(label+'-make-lint.log')).open('x') as log:
            codes[label]=subprocess.run(command,cwd=cwd,stdout=log,stderr=subprocess.STDOUT).returncode
    receipt={'command':command,'snapshot':str(snapshot),'inputs':len(names),
             'python_files':sum(n.endswith('.py') for n in names),'exit_codes':codes,
             'excluded':'Immutable artifacts/results and unrelated untracked directories; every lint input is inventoried.'}
    save('lint-receipt.json',receipt)
    assert all(digest(ROOT/name)==value for name,value in inventory.items())
    print(json.dumps(receipt,indent=2))
    raise SystemExit(codes['isolated'])
elif mode=='test':
    inventory=json.loads((OUT/'candidate-inputs.json').read_text())
    assert all(digest(ROOT/name)==value for name,value in inventory.items())
    command=['make','test']
    with (OUT/'full-suite.log').open('x') as log:
        code=subprocess.run(command,cwd=ROOT,stdout=log,stderr=subprocess.STDOUT).returncode
    assert all(digest(ROOT/name)==value for name,value in inventory.items())
    receipt={'command':command,'exit_code':code,'log_sha256':digest(OUT/'full-suite.log'),
             'candidate_inputs':inventory,
             'python':subprocess.check_output([str(ROOT/'.venv/bin/python'),'--version'],text=True).strip()}
    save('full-suite-receipt.json',receipt)
    print(json.dumps(receipt,indent=2))
    raise SystemExit(code)
else:
    raise ValueError(mode)

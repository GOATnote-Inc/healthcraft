import json,os,struct,sys,time
from dataclasses import asdict
from pathlib import Path
sys.path.insert(0,'/Users/kiteboard/healthcraft/src')
from healthcraft.reconciliation.casebook import load_cases
from healthcraft.reconciliation.controller import PilotSettings,command_format_identity
from healthcraft.reconciliation.model_case import run_model_case

def partial_frame(connection,public,output):
    (output/'started.json').write_text(json.dumps({'partial_frame':True}))
    # Simulate a worker suspended mid-write: readable header, incomplete body.
    os.write(connection.fileno(),struct.pack('!i',10000)+b'prefix')
    time.sleep(30)

if __name__=='__main__':
    config={'model':'offline-fixture:latest','expected_digest':'a'*64,'expected_runtime':'fixture','settings':asdict(PilotSettings(attempt_timeout_seconds=0.5)),'command_format':command_format_identity()}
    outcome=run_model_case(load_cases()[0],config,Path(sys.argv[1]),worker_target=partial_frame)
    print(json.dumps(outcome))

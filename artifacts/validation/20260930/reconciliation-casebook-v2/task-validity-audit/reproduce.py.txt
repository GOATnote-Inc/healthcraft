selected = set((ROOT/'src/healthcraft').rglob('*.py'))

def hashes():
    return {str(p.relative_to(ROOT)):hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(selected)}
def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT, text=True).strip()
provenance={'python':platform.python_version(),'git_head':git('rev-parse','HEAD'),'git_status_porcelain':git('status','--porcelain'),'source_hashes_before':hashes(),'network':'socket connect disabled; no model/provider calls'}
(OUT/'source-before.json').write_text(json.dumps(provenance,indent=2)+'\n')
schemas={x['name']:x['parameters'] for x in json.loads((ROOT/'configs/mcp-tools.json').read_text())['tools']}
base=WorldSeeder(seed=42).seed_world(ROOT/'configs/world/mercy_point_v1.yaml')
tasks={k:load_task(ROOT/v) for k,v in TASKS.items()}
def fresh(task_id):
    task=tasks[task_id]
    world=base.snapshot()
    instant=datetime.fromisoformat(task.initial_state['time'].replace('Z','+00:00'))
    world.advance_time(int((instant-world.timestamp).total_seconds()/60))
    target=inject_task_patient(world,task.id,task.patient,task.initial_state)
    other=inject_task_patient(world,'PEER-AUDIT-UNRELATED',{'age':30,'sex':'F','chief_complaint':'Synthetic unrelated record'},task.initial_state)
    return world,target,other

def run_case(label,task_id,actions,advance=0):
    world,target,other=fresh(task_id)
    world.advance_time(advance)
    rec=ExecutionRecorder(create_server(world),world)
    for name,params in actions(target,other):
        Draft202012Validator(schemas[name]).validate(params)
        rec.call(name,params)
    traj={'task_id':task_id,'turns':[],'criteria_results':[]}
    for call in rec.calls:
        traj['turns'].append({'role':'assistant','content':'','tool_calls':[{'id':call['id'],'name':call['name'],'arguments':call['params']}]})
        traj['turns'].append({'role':'tool','tool_call_id':call['id'],'content':json.dumps(call['response'],allow_nan=False)})
    channels={}
    for ch in ['v8','v9','v10','v11']:
        live=evaluate_task(_apply_overlay_to_task(tasks[task_id],ch),{},world,rubric_channel=ch)
        replay=replay_from_trajectory(traj,tasks[task_id],ch)
        channels[ch]={'live':{c.criterion_id:dataclasses.asdict(c) for c in live.criteria_results},'replay':{c.criterion_id:dataclasses.asdict(c) for c in replay.criteria_results},'whole_task_passed':live.passed}
    return {'id':label,'task_id':task_id,'target':target,'other':other,'task_setting_time':tasks[task_id].initial_state['time'],'world_time':world.timestamp.isoformat(),'advance_after_setting_minutes':advance,'calls':rec.calls,'audit':json.loads(json.dumps([dataclasses.asdict(x) for x in world.audit_log],default=str)),'channels':channels}

cases=[]

cases.append(run_case('scj-acetaminophen-at-setting','SCJ-012',order('target')))

cases.append(run_case('scj-nonmedication-control','SCJ-012',order('target','lab')))
cases.append(run_case('ir001-bed-resources','IR-001',lambda t,o:[('checkResourceAvailability',{'resource_type':'bed'})]))
cases.append(run_case('ir002-other-patient-and-unrelated-note','IR-002',lambda t,o:[('searchEncounters',{'patient_id':o['patient_id']}),('getPatientHistory',{'patient_id':o['patient_id']}),('updateEncounter',{'encounter_id':o['encounter_id'],'notes':'Synthetic unrelated note; target history was not reviewed.'}),('getEncounterDetails',{'encounter_id':o['encounter_id']})]))
cases.append(run_case('ir002-target-empty-search','IR-002',lambda t,o:[('searchEncounters',{'patient_id':t['patient_id'],'date_from':'2000-01-01T00:00:00Z','date_to':'2000-01-02T00:00:00Z'})]))
cases.append(run_case('ir002-target-no-linked-history','IR-002',lambda t,o:[('getPatientHistory',{'patient_id':t['patient_id']})]))

world,target,other=fresh('IR-002')

reachability={'target':target,'calls':r.calls,'source_prior_visits':tasks['IR-002'].patient.get('prior_ed_visits_30_days')}

challenge=[]
for c in json.loads((ROOT/'configs/evaluation/grader_challenges_v1.json').read_text())['cases']:
    if c['task_id'] in tasks:

            result=replay_from_trajectory(c['trajectory'],tasks[c['task_id']],ch)
            selected_result=next(x for x in result.criteria_results if x.criterion_id==c['criterion_id'])
            challenge.append({'case_id':c['id'],'task_id':c['task_id'],'criterion_id':c['criterion_id'],'channel':ch,'expected_satisfied':c['expected_satisfied'],'observed':selected_result.satisfied,'rationale':c['rationale']})

def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False,
                      allow_nan=False).encode('utf-8')


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def pid(case, number):
    return f'PAT-2{case:03d}{number:04X}'


def eid(case, number):
    return f'ENC-2{case:03d}{number:04X}'


def source(encounter, collection, position, **facts):
    return (encounter, collection, position, facts)

def materialize(case):
    number = case['number']
    scenario_id = f'synthetic-ed-reconciliation/v2/REC2-{number:03d}'
    first, last = case['name']
    scenario = {
        'schema_version': 'healthcraft-reconciliation-scenario/v2',
        'id': scenario_id,
        'clock': case['clock'],
        'target': {'patient_id': pid(number, 1), 'encounter_id': eid(number, 1)},
        'patients': [
            {'id': pid(number, person), 'mrn': f'SYNTHETIC-REC2-{number:03d}-{person}',
             'first_name': first, 'last_name': last, 'date_of_birth': None, 'sex': '',
             'prior_visit_ids': [eid(number, prior) for prior in priors]}
            for person, priors in case['patients']
        ],
        'encounters': [
            {'id': eid(number, encounter), 'patient_id': pid(number, owner),
             'chief_complaint': 'Synthetic source reconciliation', 'arrival_time': arrival,
             'patient_data': {}}
            for encounter, owner, arrival in case['encounters']
        ],
    }
    lookup = {row['id']: row for row in scenario['encounters']}
    ledger = []
    for encounter, collection, position, facts in case['rows']:
        encounter_row = lookup[eid(number, encounter)]
        data = encounter_row['patient_data']
        if collection == 'imaging_pending':
            data.setdefault(collection, {})[position] = deepcopy(facts)
        else:
            values = data.setdefault(collection, [])
            assert len(values) == position
            values.append(deepcopy(facts))
        # This is the explicitly declared shared authoring ledger, not candidate output.
        ledger.append({
            'source_id': facts['source_id'], 'patient_id': encounter_row['patient_id'],
            'encounter_id': encounter_row['id'], 'source_collection': collection,
            'source_path': '/' + collection + '/' + str(position).replace('~', '~0').replace('/', '~1'),
            'source': deepcopy(facts),
        })
    expected = {
        'schema_version': 'healthcraft-reconciliation-expectations/v2',
        'scenario_id': scenario_id, 'scenario_sha256': sha(canonical(scenario)),
        'target': {'patient_id': pid(number, 1), 'encounter_id': eid(number, 1)},
        'sources': ledger,
        'observation_source_ids': deepcopy(case['observations']),
        'unresolved_conflicts': deepcopy(case['conflicts']),
        'scope_exclusions': [
            {'source_id': source_id, 'patient_id': pid(number, owner),
             'encounter_id': eid(number, encounter), 'reason': reason}
            for source_id, owner, encounter, reason in case['exclusions']
        ],
    }
    return scenario, expected


def write(path, value):
    raw = (json.dumps(value, ensure_ascii=False, indent=2, allow_nan=False) + '\n').encode('utf-8')
    with path.open('xb') as stream:
        stream.write(raw)
    return sha(raw)


def main():
    ROOT.mkdir()
    casebook = {
        'schema_version': 'healthcraft-reconciliation-casebook/v2',
        'id': 'source-reconciliation-development/v2', 'exposure': 'development',
        'label_status': 'engineering_authored_independent_review_pending',
        'cases': [], 'files': {},
    }
    counts = []
    for case in CASES:
        identifier = f"REC2-{case['number']:03d}"
        directory = ROOT / identifier
        directory.mkdir()
        scenario, expected = materialize(case)
        for name, value in [('scenario', scenario), ('expectations', expected)]:
            relative = f'{identifier}/{name}.json'
            casebook['files'][relative] = write(ROOT / relative, value)
        casebook['cases'].append({
            'case_id': identifier, 'scenario_family_id': case['family'],
            'scenario': f'{identifier}/scenario.json', 'scenario_sha256': sha(canonical(scenario)),
            'expectations': f'{identifier}/expectations.json',
            'expectations_sha256': sha(canonical(expected)),
            'designated_control': case['control'],
        })
        counts.append({'case_id': identifier, 'patients': len(scenario['patients']),
                       'encounters': len(scenario['encounters']), 'source_rows': len(expected['sources']),
                       'target_rows': len(expected['observation_source_ids']),
                       'excluded_rows': len(expected['scope_exclusions']),
                       'conflict_clusters': len(expected['unresolved_conflicts']),
                       'designated_control': case['control']})
    raw_hash = write(ROOT / 'casebook.json', casebook)
    receipt = {
        'schema_version': 'healthcraft-reconciliation-development-authoring/v1',
        'status': 'authored_independent_review_pending',
        'casebook_canonical_sha256': sha(canonical(casebook)),
        'casebook_file_sha256': raw_hash,
        'files': casebook['files'], 'counts': counts,
        'total_source_rows': sum(row['source_rows'] for row in counts),
        'authoring_method': 'An explicit shared authoring ledger supplies raw record facts and ownership for scenario and expectations serialization. Target inclusion, unresolved-conflict membership and scope exclusions are separately manually enumerated. No candidate fixture/controller/oracle/diagnostic imports or executions were used.',
        'limitations': [
            'Shared authoring is not independent labeling or independent review.',
            'Development cases are newly authored but exposed, not held out.',
            'No human participants, clinical labels, treatments recommended, models or task executions.',
            'designated_control declares a future engineering trace, not an observed outcome.',
            'Source timestamps remain literal strings or null; repeated reports are not adjudicated clinical facts.',
        ],
        'authoring_script_sha256': sha(Path(__file__).read_bytes()),
    }
    write(RECEIPTS / 'authoring-receipt.json', receipt)
    print(json.dumps({'counts': counts, 'total_source_rows': receipt['total_source_rows'],
                      'casebook_canonical_sha256': receipt['casebook_canonical_sha256'],
                      'casebook_file_sha256': raw_hash}, indent=2))


if __name__ == '__main__':
    main()

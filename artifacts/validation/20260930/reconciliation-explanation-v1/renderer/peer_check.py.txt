class Document(HTMLParser):
    def __init__(self, html):
        super().__init__(convert_charrefs=True)
        self.tags = []
        self.stack = []
        self.parts = []
        self.counts = {}
        self.feed(html)
        self.close()

    def handle_starttag(self, tag, attrs):
        attrs = dict(attrs)
        self.tags.append((tag, attrs))
        if tag not in {'meta', 'br', 'hr', 'input', 'img', 'link'}:
            self.stack.append((tag, attrs))
        if 'data-count' in attrs:
            self.counts[attrs['data-count']] = ''

    def handle_endtag(self, tag):
        for i in range(len(self.stack)-1, -1, -1):
            if self.stack[i][0] == tag:
                del self.stack[i:]
                break

    def handle_data(self, data):
        if not any(tag == 'style' for tag, _ in self.stack):
            self.parts.append(data)
        for _, attrs in self.stack:
            if 'data-count' in attrs:
                self.counts[attrs['data-count']] += data

    @property
    def text(self):
        return ' '.join(self.parts)

    @property
    def inert(self):
        blocked = {'script', 'iframe', 'object', 'embed', 'img', 'svg', 'math', 'a',
                   'link', 'form', 'input', 'button', 'video', 'audio', 'base'}
        return not any(tag in blocked or any(k.startswith('on') or k in {'href', 'src',
            'srcdoc', 'action', 'formaction'} for k in attrs) for tag, attrs in self.tags)


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def refs(value):
    if isinstance(value, dict):
        if 'document' in value and 'pointer' in value:
            yield value
        for child in value.values():
            yield from refs(child)
    elif isinstance(value, list):
        for child in value:
            yield from refs(child)

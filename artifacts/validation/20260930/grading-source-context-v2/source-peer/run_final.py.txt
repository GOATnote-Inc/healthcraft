"""Run source-bound independent static probes only; never model/browser/network."""
import argparse
import datetime
import hashlib
import json
import subprocess
from pathlib import Path

ROOT = Path('/Users/kiteboard/healthcraft')
HERE = Path(__file__).resolve().parent
FILES = [
    'src/healthcraft/reconciliation/explanation_report.py',
    'scripts/explain_reconciliation.py',
    'tests/test_reconciliation/test_explanation_sources.py',
    'tests/test_scripts/test_explanation_source_bundle.py',
    'src/healthcraft/reconciliation/diagnostics.py',
    'src/healthcraft/reconciliation/oracle.py',
]

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def source():
    return {f:sha(ROOT/f) for f in FILES}

p = argparse.ArgumentParser()
p.add_argument('--renderer-sha', required=True)
p.add_argument('--renderer-test-sha', required=True)
a = p.parse_args()
expected = {
    FILES[0]:a.renderer_sha,
    FILES[1]:'a1f9a89d9773dc537b42e42e7845f0cc289ccd85970e1cf35d50faa332687432',
    FILES[2]:a.renderer_test_sha,
    FILES[3]:'d92847674adee750c39f28f03285615c1f9cbcb2789c0b5ac969a57b05c05644',
}
before = source()
assert all(before[k] == v for k,v in expected.items())
initial = json.loads((HERE/'source-before.json').read_text())
runs = []
for version, python in [
    ('3.10','/private/tmp/healthcraft-check-py310/bin/python'),
    ('3.12','/private/tmp/healthcraft-check-py312/bin/python'),
    ('3.14',str(ROOT/'.venv/bin/python')),
]:
    cmd = [python,'-m','pytest',str(HERE/'test_source_context_peer.py'),'-q']
    env = {'PATH':'/bin:/usr/bin','PYTHONPATH':str(ROOT/'src'),'PYTEST_DISABLE_PLUGIN_AUTOLOAD':'1'}
    result = subprocess.run(cmd,cwd=ROOT,env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
    log = HERE/('final-py'+version.replace('.','')+'.log')
    with log.open('x') as f:
        f.write(result.stdout)
    runs.append({'python':version,'command':cmd,'environment':env,'exit_code':result.returncode,
                 'log':log.name,'log_sha256':sha(log)})
    print(version,result.stdout.strip())
    assert result.returncode == 0

after = source()
assert before == after
unchanged = all(sha(ROOT/k) == v for k,v in initial['saved_bundle_files'].items())
assert unchanged

def make_world():
    world = WorldState(CLOCK)
    for p in ('A','B'):
        patient = Patient(id=f'PAT-{p*8}',entity_type=EntityType.PATIENT,created_at=CLOCK,updated_at=CLOCK,mrn=f'SYNTHETIC-{p}',first_name='Rowan',last_name='Example',dob=None,prior_visit_ids=('ENC-CCCCCCCC',) if p=='A' else ())
        world.put_entity('patient',patient.id,patient)
    for e,p,source in [('A','A',SOURCE),('B','B',{'treatments_given':[{'source_id':'SRC-B01','item':'synthetic_medication_C','reported_status':'administered','event_time':'2026-09-30T14:05:00Z'}]}),('C','A',{'treatments_given':[{'source_id':'SRC-A07','item':'synthetic_medication_A','reported_status':'administered','event_time':None}]})]:
        imaging=project_imaging(source,task_id='SYNTHETIC-EHR-V1')
        enc=Encounter(id=f'ENC-{e*8}',entity_type=EntityType.ENCOUNTER,created_at=CLOCK,updated_at=CLOCK,patient_id=f'PAT-{p*8}',chief_complaint='Synthetic source reconciliation',esi_level=None,arrival_time=CLOCK if e in 'AB' else None,authored_care=project_authored_care(source),imaging=imaging.records,clinical_notes=(('Source context','Wholly synthetic, no diagnosis or treatment selection requested.'),))
        world.put_entity('encounter',enc.id,enc)
    return world

def call(rec,name,params):
    jsonschema.validate(params,SCHEMAS[name])
    return rec.call(name,params)

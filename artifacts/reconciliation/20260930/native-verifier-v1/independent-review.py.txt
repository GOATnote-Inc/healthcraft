def require(name, condition, detail=None):
    checks.append({'check':name,'passed':bool(condition), 'detail':detail})
    if not condition:

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def pairs(items):
    d = {}
    for key, value in items:
        if key in d: raise ValueError('Duplicate JSON key: ' + key)
        d[key] = value
    return d

def number(value):
    n = float(value)
    if not math.isfinite(n): raise ValueError('Nonfinite JSON number')
    return n

def parse(text):
    return json.loads(text, object_pairs_hook=pairs, parse_float=number, parse_constant=number)

def load(path):
    return parse(path.read_text())

def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False, allow_nan=False)

class ReportParser(HTMLParser):
    def __init__(self):
        super().__init__();self.links=[];self.active=[];self.table_rows=0
    def handle_starttag(self,tag,attrs):
        if tag=='tr':self.table_rows+=1
        if tag in ('script','iframe','object','embed'):self.active.append(tag)
        for name,value in attrs:
            if name.lower().startswith('on'):self.active.append(name)
            if name in ('href','src'):self.links.append(value)

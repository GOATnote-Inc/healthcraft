def write(path,value):
    with path.open('x') as f: json.dump(value,f,indent=2,sort_keys=True,allow_nan=False); f.write('\n')
def command(argv,where,*,env=None,timeout=120,check=True):
    started=time.monotonic()
    try:
        p=subprocess.run(argv,cwd=ROOT,env=env,capture_output=True,text=True,timeout=timeout)
        receipt={'argv':argv,'returncode':p.returncode,'stdout':p.stdout,'stderr':p.stderr,'elapsed_seconds':time.monotonic()-started}
    except subprocess.TimeoutExpired as e:
        receipt={'argv':argv,'returncode':None,'error':'TimeoutExpired','stdout':str(e.stdout),'stderr':str(e.stderr),'elapsed_seconds':time.monotonic()-started}
    except OSError as e:
        receipt={'argv':argv,'returncode':None,'error':type(e).__name__,'stdout':'','stderr':str(e),'elapsed_seconds':time.monotonic()-started}
    write(where,receipt)
    if check and receipt['returncode']!=0: raise RuntimeError(f'Command failed; see {where.name}')
    return receipt

def source_check():
    differences=[]

        try: actual=hashlib.sha256((ROOT/relative).read_bytes()).hexdigest()
        except OSError: actual=None
        if actual!=expected: differences.append({'path':relative,'expected':expected,'actual':actual})
    return {'matched':not differences,'files_checked':len(HOST_INPUTS),'differences':differences}

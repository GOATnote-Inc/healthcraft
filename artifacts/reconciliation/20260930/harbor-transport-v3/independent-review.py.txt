def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
def strict(s):
    def obj(items):
        value = {}
        for key, val in items:
            if key in value: raise ValueError('duplicate key')
            value[key] = val
        return value
    def number(s):
        value = float(s)
        if not math.isfinite(value): raise ValueError('nonfinite')
        return value
    return json.loads(s, object_pairs_hook=obj, parse_constant=number, parse_float=number)
def read(p): return strict(p.read_text())
def canonical(v): return json.dumps(v,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False)
def identical(a,b): return canonical(a)==canonical(b)
def lines(p): return [strict(s) for s in p.read_text().splitlines() if s]
def digest_value(v): return hashlib.sha256(canonical(v).encode()).hexdigest()

def check(key, value): checks[key] = bool(value)

from pathlib import Path
import datetime, hashlib, json, subprocess, time
candidate=Path('/private/tmp/healthcraft-care-final-ydju6d2l')
out=Path('/Users/kiteboard/healthcraft/artifacts/local-care-probe/20260930/source-reading-v1')
out.mkdir(parents=True,exist_ok=False)
command=['/Users/kiteboard/healthcraft/.venv/bin/python','scripts/local_care_probe.py','--output',str(out/'attempts.json'),'--timeout','45','--max-output-tokens','768']
env={'PATH':'/usr/bin:/bin','PYTHONUNBUFFERED':'1'}
receipt={'kind':'bounded-local-process/v1','command':command,'cwd':str(candidate),'environment':env,'outer_timeout_seconds':320,'status':'started','started_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'runner_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
receipt_path=out/'process.json'
with receipt_path.open('x') as f:json.dump(receipt,f,indent=2)
started=time.monotonic()
try:
 with (out/'process.log').open('xb') as log:
  result=subprocess.run(command,cwd=candidate,env=env,stdout=log,stderr=subprocess.STDOUT,timeout=320)
 receipt.update(status='finished',exit_code=result.returncode)
except Exception as exc:
 receipt.update(status='process_error',error=f'{type(exc).__name__}: {exc}',exit_code=None)
receipt.update(duration_seconds=round(time.monotonic()-started,3),ended_at=datetime.datetime.now(datetime.timezone.utc).isoformat())
receipt_path.write_text(json.dumps(receipt,indent=2)+'\n')
print(json.dumps(receipt))
if (out/'attempts.json').exists():
 r=json.loads((out/'attempts.json').read_text())
 print(json.dumps({k:r.get(k) for k in ('diagnostic_status','execution_counts','provenance_stable','duration_seconds')}))
 for a in r.get('attempts',[]):print(json.dumps({k:a.get(k) for k in ('model','status','completion','source_reading','errors','duration_seconds')}))
raise SystemExit(receipt['exit_code'] if receipt['exit_code'] is not None else 2)
